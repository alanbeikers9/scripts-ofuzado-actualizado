local function normalizarURL(url)
    if type(url) ~= "string" then return "", "invalido" end
    url = url:gsub("^%s+", ""):gsub("%s+$", "")
    if url == "" then return "", "vacio" end

    for _, regla in ipairs(PATRONES_URL) do
        local c1, c2, c3, c4 = url:match(regla.patron)
        if c1 ~= nil or not regla.patron:find("%(") then
            if not (regla.excluir and c1 and regla.excluir(c1)) then
                if regla.limpiar and c4 then
                    c4 = c4:gsub("%?.*$", "")
                end
                if regla.formato then
                    local ok, resultado = pcall(string.format, regla.formato, c1, c2, c3, c4)
                    if ok and resultado then return resultado, regla.nombre end
                else
                    return url, regla.nombre
                end
            end
        end
    end

    return url, "desconocida (falta https://?)"
end

-- =====================================================================
--  GENERADOR
-- =====================================================================
local function hashURL(url)
    local hash = 5381
    for i = 1, #url do
        hash = (hash * 33 + url:byte(i)) % 2147483647
    end
    return hash
end

local function nombreGuard(url)
    return "Loadstringer_" .. tostring(hashURL(url))
end

local function escaparParaLua(texto)
    return (texto:gsub("\\", "\\\\"):gsub('"', '\\"'))
end

local function generarLoadstring()
    if estado.urlOriginal == "" then
        return "-- Pega una URL arriba para comenzar"
    end

    local urlFinal = escaparParaLua(estado.urlNormalizada)

    local httpCall = estado.usarTrue
        and string.format('game:HttpGet("%s", true)', urlFinal)
        or  string.format('game:HttpGet("%s")', urlFinal)

    local cuerpo = "loadstring(" .. httpCall .. ")"

    if estado.soloFuncion then
        if estado.conComentario then
            return "--[Loadstringer] " .. estado.urlNormalizada .. "\n" .. cuerpo
        end
        return cuerpo
    end

    cuerpo = cuerpo .. "()"

    if estado.usarGetgenv then
        local guard = nombreGuard(estado.urlNormalizada)
        cuerpo = string.format(
            "if not getgenv().%s then\n    getgenv().%s = true\n    %s\nend",
            guard, guard, cuerpo:gsub("\n", "\n    ")
        )
    end

    if estado.usarPcall then
        if estado.pcallVerbose then
            local indentado = cuerpo:gsub("\n", "\n    ")
            cuerpo = string.format(
                "local ok, err = pcall(function()\n    %s\nend)\nif not ok then warn(\"[Loadstringer] Error:\", err) end",
                indentado
            )
        else
            cuerpo = "pcall(function() " .. cuerpo .. " end)"
        end
    end

    if estado.conComentario then
        cuerpo = "--[Loadstringer] " .. estado.urlNormalizada .. "\n" .. cuerpo
    end

    return cuerpo
end

--  HISTORIAL
-- =====================================================================
local HISTORIAL_MAX  = 10
local HISTORIAL_RUTA = "Loadstringer/historial.json"
local historialCache = nil

local function asegurarCarpetaHistorial()
    if not fsDisponible() then return end
    if type(isfolder) == "function" and type(makefolder) == "function" then
        if not isfolder("Loadstringer") then pcall(makefolder, "Loadstringer") end
    end
end

local function cargarHistorial()
    if historialCache then return historialCache end
    historialCache = {}
    if not fsDisponible() then return historialCache end
    if type(isfile) ~= "function" or not isfile(HISTORIAL_RUTA) then return historialCache end

    local ok, texto = pcall(readfile, HISTORIAL_RUTA)
    if not ok or not texto then return historialCache end

    local okDec, datos = pcall(function() return HttpService:JSONDecode(texto) end)
    if okDec and type(datos) == "table" and type(datos.urls) == "table" then
        for _, u in ipairs(datos.urls) do
            if type(u) == "string" and u ~= "" and #historialCache < HISTORIAL_MAX then
                table.insert(historialCache, u)
            end
        end
    end
    return historialCache
end

local function guardarHistorial()
    if not fsDisponible() or type(writefile) ~= "function" then return end
    asegurarCarpetaHistorial()
    local datos = { urls = historialCache or {} }
    pcall(function()
        writefile(HISTORIAL_RUTA, HttpService:JSONEncode(datos))
    end)
end

local function agregarAlHistorial(url)
    if type(url) ~= "string" or url == "" then return cargarHistorial() end
    local h = cargarHistorial()
    for i = #h, 1, -1 do
        if h[i] == url then table.remove(h, i) end
    end
    table.insert(h, 1, url)
    while #h > HISTORIAL_MAX do table.remove(h) end
    guardarHistorial()
    return h
end

local function limpiarHistorial()
    historialCache = {}
    guardarHistorial()
    return historialCache
end

local function formatearTamano(n)
    if n < 1000 then return n .. " chars" end
    if n < 1000000 then return string.format("%.1fk chars", n / 1000) end
    return string.format("%.1fM chars", n / 1000000)
end

local secPrev = Tab:CrearSeccion("Vista previa", {
    Colapsable = true,
    ColapsadaPorDefecto = true,
})

secPrev:Boton({
    Texto    = "Previsualizar URL",
    Tooltip  = "Descarga y muestra el contenido sin ejecutarlo",
    Callback = function()
        if estado.urlOriginal == "" then
            AlanUI:Notificar({ Titulo = "Aviso", Texto = "Falta URL", Tipo = "Warning", Duracion = 2 })
            return
        end
        if estado.urlNormalizada == "" then
            AlanUI:Notificar({ Titulo = "Error", Texto = "URL invalida", Tipo = "Error", Duracion = 3 })
            return
        end

        local url   = estado.urlNormalizada
        local cache = cachePreview[url]
        if cache and (os.clock() - cache.t) < PREVIEW_TTL then
            parrafoPreview:Set(cache.content)
            AlanUI:Notificar({ Titulo = "Aviso", Texto = "Usando cache (60s)", Tipo = "Info", Duracion = 2 })
            return
        end

        parrafoPreview:Set("Descargando " .. url .. " ...")
        AlanUI:Notificar({ Titulo = "Descargando", Texto = "Espera un momento...", Tipo = "Info", Duracion = 2 })

        task.spawn(function()
            local ok, resultado = pcall(function()
                return game:HttpGet(url, true)
            end)

            if not ok then
                parrafoPreview:Set("Error al descargar:\n\n" .. tostring(resultado))
                AlanUI:Notificar({ Titulo = "Error", Texto = "No se pudo descargar", Tipo = "Error", Duracion = 3 })
                return
            end
            if resultado == nil then
                parrafoPreview:Set("La URL devolvio nil")
                return
            end
            if type(resultado) ~= "string" then
                resultado = tostring(resultado)
            end
            resultado = resultado:gsub("\r\n", "\n"):gsub("\r", "\n")

            local contenidoMostrar = resultado
            if #contenidoMostrar > PREVIEW_LIMITE then
                contenidoMostrar = contenidoMostrar:sub(1, PREVIEW_LIMITE)
                    .. "\n\n[Truncado: " .. #resultado .. " caracteres totales]"
            end

            cachePreview[url] = { content = contenidoMostrar, t = os.clock() }
            parrafoPreview:Set(contenidoMostrar)

            local h = agregarAlHistorial(url)
            if dropdownHistorial and dropdownHistorial.ActualizarOpciones then 
                dropdownHistorial:ActualizarOpciones(h) 
            end

            AlanUI:Notificar({
                Titulo = "Descargado",
                Texto  = #resultado .. " caracteres",
                Tipo   = "Success",
                Duracion = 2,
            })
        end)
    end,
})

-- =====================================================================
--  FUNCION CENTRAL
-- =====================================================================
actualizarTodo = function()
    local normalizada, tipo = normalizarURL(estado.urlOriginal)
    estado.urlNormalizada = normalizada
    estado.tipoDetectado  = tipo

    if lblTipoURL and lblTipoURL.Set then lblTipoURL:Set(tipo) end

    ultimoResultado = generarLoadstring()
    if parrafoResultado and parrafoResultado.Set then parrafoResultado:Set(ultimoResultado) end

    if lblStats and lblStats.Set then
        local nChars = #ultimoResultado
        local _, nSaltos = ultimoResultado:gsub("\n", "")
        local nLineas = nSaltos + 1
        lblStats:Set(formatearTamano(nChars) .. " | " .. nLineas .. " lineas")
    end
end

task.defer(function()
    if dropdownHistorial then
        historialCache = nil
        local opts = cargarHistorial()
        if dropdownHistorial.ActualizarOpciones then
            dropdownHistorial:ActualizarOpciones(opts)
        end
    end
    actualizarTodo()
end)
